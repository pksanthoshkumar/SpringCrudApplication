To model this Spring Boot application for processing SOAP requests, you have two main options:

Implementing a SOAP-based Web Service: Since your request is a SOAP XML format, you could create a SOAP web service in Spring Boot using the Spring Web Services module (spring-ws). This is the traditional way of handling SOAP requests.

REST Application with SOAP Support: While REST applications are not natively designed to handle SOAP requests, they can process SOAP messages if you parse the incoming SOAP request manually or use a filter to intercept SOAP requests and convert them to a format that can be handled by your REST controllers.

For simplicity and to better align with Spring's native support, I recommend using Spring Web Services rather than REST, as it simplifies handling SOAP requests and responses.

package com.example.chequereading;

import org.springframework.boot.web.servlet.ServletRegistrationBean;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.ws.transport.http.MessageDispatcherServlet;
import org.springframework.ws.wsdl.wsdl11.DefaultWsdl11Definition;
import org.springframework.xml.xsd.SimpleXsdSchema;
import org.springframework.xml.xsd.XsdSchema;

@Configuration
public class WebServiceConfig {

    @Bean
    public ServletRegistrationBean<MessageDispatcherServlet> messageDispatcherServlet(ApplicationContext context) {
        MessageDispatcherServlet servlet = new MessageDispatcherServlet();
        servlet.setApplicationContext(context);
        servlet.setTransformWsdlLocations(true);
        return new ServletRegistrationBean<>(servlet, "/ws/*");
    }

    @Bean(name = "cheque")
    public DefaultWsdl11Definition defaultWsdl11Definition(XsdSchema chequeSchema) {
        DefaultWsdl11Definition definition = new DefaultWsdl11Definition();
        definition.setPortTypeName("ChequePort");
        definition.setTargetNamespace("http://www.tibco.com/schemas/CheckImage/MonetaryInstrument");
        definition.setLocationUri("/ws");
        definition.setSchema(chequeSchema);
        return definition;
    }

    @Bean
    public XsdSchema chequeSchema() {
        return new SimpleXsdSchema(new ClassPathResource("ChequeRequest.xsd"));
    }
}
