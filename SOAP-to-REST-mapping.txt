package com.example.chequereadingapp.controller;

import com.example.chequereadingapp.domain.CheckImageRequest;
import com.example.chequereadingapp.service.CheckImageService;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.web.bind.annotation.*;
import jakarta.xml.bind.JAXBContext;
import jakarta.xml.bind.JAXBException;
import jakarta.xml.bind.Unmarshaller;
import javax.xml.transform.stream.StreamSource;
import java.io.StringReader;

@RestController
@RequestMapping("/soap")
public class CheckImageController {

    @Autowired
    private CheckImageService checkImageService;

    @PostMapping(value = "/getCheckImage", consumes = "application/xml", produces = "application/xml")
    public String getCheckImage(@RequestBody String xmlRequest) throws JAXBException {
        JAXBContext jaxbContext = JAXBContext.newInstance(CheckImageRequest.class);
        Unmarshaller unmarshaller = jaxbContext.createUnmarshaller();

        CheckImageRequest request = (CheckImageRequest) unmarshaller.unmarshal(new StreamSource(new StringReader(xmlRequest)));
        return checkImageService.getCheckImage(request);
    }
}
