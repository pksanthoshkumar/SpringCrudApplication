    /**
     * Generic retry mechanism.
     *
     * @param action        The action to retry (a functional interface).
     * @param maxRetries    The maximum number of retry attempts.
     * @param waitTimeInMs  The delay between retries in milliseconds.
     * @param <T>           The return type of the action.
     * @return The result of the action if successful, or null if all retries fail.
     */
    private <T> T retry(Retryable<T> action, int maxRetries, long waitTimeInMs) {
        int retryCount = 0;
        while (retryCount < maxRetries) {
            try {
                return action.execute();
            } catch (Exception e) {
                System.err.println("Attempt " + (retryCount + 1) + " failed: " + e.getMessage());
            }

            retryCount++;
            try {
                Thread.sleep(waitTimeInMs);
            } catch (InterruptedException ex) {
                Thread.currentThread().interrupt();
                System.err.println("Retry interrupted.");
                break;
            }
        }
        return null;
    }

    @FunctionalInterface
    public interface Retryable<T> {
        T execute() throws Exception;
    }
}
