import com.example.demo.domain.MonetaryInstrumentRequest;
import com.example.demo.service.CheckImageService;
import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;
import org.mockito.Mockito;
import org.springframework.beans.factory.annotation.Autowired;

import org.springframework.boot.test.autoconfigure.web.servlet.WebMvcTest;
import org.springframework.boot.test.mock.mockito.MockBean;
import org.springframework.http.MediaType;
import org.springframework.test.web.servlet.MockMvc;

import static org.mockito.ArgumentMatchers.any;
import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.post;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.content;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.status;

@WebMvcTest(ImageRetrievalController.class)
class ImageRetrievalControllerTest {

    @Autowired
    private MockMvc mockMvc;

    @MockBean
    private BookImageService BookImageService;

    private String validXmlRequest;
    private String invalidXmlRequest;

    @BeforeEach
    void setUp() {
        // Example of a valid XML request
        validXmlRequest = """
            <XMLReqq>
                <Header>
                    <RequestingApp>SampleApp</RequestingApp>
                </Header>
                <BookNbr>
                    <Low>12345</Low>
                    <High>67890</High>
                </BookNbr>
                <TitleNbr>987654321</TitleNbr>
            </XMLReqq>
            """;

        // Example of an invalid XML request
        invalidXmlRequest = """
            <InvalidXML>
                <BadTag>12345</BadTag>
            </InvalidXML>
            """;
    }

    @Test
    void testGetBookImage_ValidRequest() throws Exception {
        // Mock the service response
        Mockito.when(BookImageService.getBookImage(any(Request.class)))
                .thenReturn("<Response>Success</Response>");

        // Perform the test
        mockMvc.perform(post("/getBookImage")
                        .contentType(MediaType.APPLICATION_XML)
                        .content(validXmlRequest))
                .andExpect(status().isOk())
                .andExpect(content().xml("<Response>Success</Response>"));

        // Verify the service interaction
        Mockito.verify(BookImageService, Mockito.times(1)).getBookImage(any(Request.class));
    }

    @Test
    void testGetBookImage_InvalidRequest() throws Exception {
        // Perform the test with an invalid XML
        mockMvc.perform(post("/getBookImage")
                        .contentType(MediaType.APPLICATION_XML)
                        .content(invalidXmlRequest))
                .andExpect(status().isOk())
                .andExpect(content().string("INVALID_SOAP_REQUEST"));

        // Verify the service was not called
        Mockito.verify(BookImageService, Mockito.never()).getBookImage(any(Request.class));
    }

    @Test
    void testGetBookImage_ExceptionHandling() throws Exception {
        // Simulate an exception in the service
        Mockito.when(BookImageService.getBookImage(any(Request.class)))
                .thenThrow(new RuntimeException("Service error"));

        // Perform the test
        mockMvc.perform(post("/getBookImage")
                        .contentType(MediaType.APPLICATION_XML)
                        .content(validXmlRequest))
                .andExpect(status().isOk())
                .andExpect(content().string("INVALID_SOAP_REQUEST"));

        // Verify the service interaction
        Mockito.verify(BookImageService, Mockito.times(1)).getBookImage(any(Request.class));
    }
}
